# Picker Performance Reports - Redesigned

## Problem with Current Reports

1. **Items Picked = 0** (join bug with item_time_tracking)
2. **Items-per-hour = 0** (cascade from items picked)
3. **Performance = 0** (not calculated)
4. **No activity breakdown** (idle not categorized)
5. **No quality metrics** (skip rates, accuracy)
6. **Shift auto-close** inflates idle time
7. **No trends** (can't compare day-to-day)

---

## Solution: End-of-Shift Summary Report

**Trigger:** When picker checks out (every shift)  
**Shows:** All four metric categories in one unified report  
**Output:** HTML email + dashboard card + printable PDF  
**Data:** Real-time queries pulling from item_time_tracking, idle_periods, activity_transitions

## Detailed SQL Queries (Fix Current Report)

### 1. Speed & Throughput (Fixed Join)

```sql
CREATE VIEW vw_picker_daily_summary AS
SELECT 
    -- Picker & Date
    ipt.picker_username,
    DATE(ipt.created_at) as work_date,
    
    -- Items & Quantity
    COUNT(DISTINCT ipt.id) as items_completed,
    SUM(ipt.quantity_picked) as total_quantity_picked,
    COUNT(DISTINCT ipt.invoice_no) as orders_completed,
    
    -- Speed (Core Metrics)
    ROUND(AVG(CASE WHEN ipt.total_item_time > 0 THEN ipt.total_item_time END)::numeric, 1) as avg_time_per_item_sec,
    ROUND((SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.quantity_picked END) * 3600.0 
            / SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.total_item_time END))::numeric, 1) as items_per_hour,
    ROUND(AVG(CASE WHEN ipt.total_item_time > 0 THEN ipt.efficiency_ratio END)::numeric, 3) as avg_efficiency_ratio,
    
    -- Time Breakdown
    ROUND(SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.total_item_time END) / 60.0::numeric, 1) as picking_total_min,
    ROUND(SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.walking_time END) / 60.0::numeric, 1) as walking_total_min,
    ROUND(SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.picking_time END) / 60.0::numeric, 1) as picking_actual_min,
    ROUND(SUM(CASE WHEN ipt.total_item_time > 0 THEN ipt.confirmation_time END) / 60.0::numeric, 1) as confirmation_total_min,
    
    -- Quality
    COUNT(CASE WHEN ipt.was_skipped THEN 1 END) as items_skipped,
    COUNT(CASE WHEN ipt.picked_correctly = false THEN 1 END) as items_with_errors,
    ROUND(100.0 * COUNT(CASE WHEN ipt.picked_correctly = true THEN 1 END) / NULLIF(COUNT(*), 0)::numeric, 1) as accuracy_pct
    
FROM item_time_tracking ipt
WHERE ipt.total_item_time > 0  -- Exclude incomplete records
GROUP BY 1, 2
ORDER BY work_date DESC, picker_username;
```

### 2. Occupancy & Utilization

```sql
CREATE VIEW vw_picker_occupancy_summary AS
SELECT 
    pat.picker_username,
    DATE(pat.activity_start) as work_date,
    
    -- Shift Duration
    s.id as shift_id,
    ROUND((EXTRACT(EPOCH FROM (s.check_out_time - s.check_in_time)) / 60)::numeric, 1) as shift_duration_min,
    
    -- Activity Time Breakdown
    ROUND(SUM(CASE WHEN pat.activity = 'break' THEN pat.activity_duration_sec ELSE 0 END) / 60.0::numeric, 1) as break_min,
    ROUND(SUM(CASE WHEN pat.activity = 'restock' THEN pat.activity_duration_sec ELSE 0 END) / 60.0::numeric, 1) as restock_min,
    ROUND(SUM(CASE WHEN pat.activity = 'assist' THEN pat.activity_duration_sec ELSE 0 END) / 60.0::numeric, 1) as assist_min,
    ROUND(SUM(CASE WHEN pat.activity = 'repacking' THEN pat.activity_duration_sec ELSE 0 END) / 60.0::numeric, 1) as repacking_min,
    
    -- Occupied Time (when orders are open)
    ROUND((EXTRACT(EPOCH FROM (MAX(CASE WHEN pat.activity = 'pick_orders' THEN pat.activity_end ELSE pat.activity_start END) 
                                  - MIN(pat.activity_start))) / 60)::numeric, 1) as occupied_total_min,
    
    -- Occupancy %
    ROUND(100.0 * 
          ROUND((EXTRACT(EPOCH FROM (MAX(CASE WHEN pat.activity = 'pick_orders' THEN pat.activity_end ELSE pat.activity_start END) 
                                      - MIN(pat.activity_start))) / 60)::numeric, 1) 
          / NULLIF(ROUND((EXTRACT(EPOCH FROM (s.check_out_time - s.check_in_time)) / 60)::numeric, 1), 0)::numeric, 1) as occupancy_pct
    
FROM picker_activity_transition pat
JOIN shifts s ON pat.shift_id = s.id
WHERE pat.picker_role = 'picker'
GROUP BY 1, 2, s.id, s.check_in_time, s.check_out_time
ORDER BY work_date DESC, picker_username;
```

### 3. Quality & Accuracy

```sql
CREATE VIEW vw_picker_quality_summary AS
SELECT 
    ipt.picker_username,
    DATE(ipt.created_at) as work_date,
    
    COUNT(*) as total_picks,
    COUNT(CASE WHEN ipt.was_skipped THEN 1 END) as skipped_count,
    ROUND(100.0 * COUNT(CASE WHEN ipt.was_skipped THEN 1 END) / NULLIF(COUNT(*), 0)::numeric, 1) as skip_rate_pct,
    
    COUNT(CASE WHEN ipt.picked_correctly = true THEN 1 END) as correct_picks,
    COUNT(CASE WHEN ipt.picked_correctly = false THEN 1 END) as error_count,
    ROUND(100.0 * COUNT(CASE WHEN ipt.picked_correctly = true THEN 1 END) / NULLIF(COUNT(*), 0)::numeric, 2) as accuracy_pct,
    
    -- Outlier Picks (>2× estimate)
    COUNT(CASE WHEN ipt.efficiency_ratio > 2.0 THEN 1 END) as very_slow_picks,
    COUNT(CASE WHEN ipt.efficiency_ratio BETWEEN 1.0 AND 2.0 THEN 1 END) as slow_picks,
    COUNT(CASE WHEN ipt.efficiency_ratio < 1.0 THEN 1 END) as fast_picks,
    
    -- Expected vs Actual
    ROUND(SUM(ipt.expected_time) / 60.0::numeric, 1) as expected_total_min,
    ROUND(SUM(ipt.total_item_time) / 60.0::numeric, 1) as actual_total_min,
    ROUND(((SUM(ipt.total_item_time) - SUM(ipt.expected_time)) / SUM(ipt.expected_time) * 100)::numeric, 1) as variance_pct
    
FROM item_time_tracking ipt
WHERE ipt.total_item_time > 0
GROUP BY 1, 2
ORDER BY work_date DESC, picker_username;
```

### 4. Weekly Trends

```sql
CREATE VIEW vw_picker_weekly_trends AS
SELECT 
    ipt.picker_username,
    DATE(ipt.created_at) as work_date,
    EXTRACT(DOW FROM ipt.created_at)::INT as day_of_week,
    
    COUNT(*) as items_daily,
    ROUND(AVG(ipt.total_item_time)::numeric, 1) as avg_time_per_item_daily,
    ROUND(AVG(ipt.efficiency_ratio)::numeric, 3) as efficiency_daily,
    ROUND(100.0 * COUNT(CASE WHEN ipt.picked_correctly = true THEN 1 END) / NULLIF(COUNT(*), 0)::numeric, 2) as accuracy_daily
    
FROM item_time_tracking ipt
WHERE ipt.total_item_time > 0
  AND DATE(ipt.created_at) >= CURRENT_DATE - INTERVAL '13 days'
GROUP BY 1, 2, 3
ORDER BY picker_username, work_date DESC;
```

## End-of-Shift Report (HTML Email + Dashboard)

### Report Generation (When Picker Checks Out)

```python
@app.route('/report/end-of-shift', methods=['POST'])
def generate_eod_report():
    """
    Called when picker checks out.
    Generates HTML report, emails it, shows on dashboard.
    """
    data = request.json
    picker_username = data['picker_username']
    shift_id = data['shift_id']
    work_date = datetime.now().date()
    
    # Query all 4 views for this picker/date
    speed = db.query("""
        SELECT * FROM vw_picker_daily_summary 
        WHERE picker_username = %s AND work_date = %s
    """, picker_username, work_date)
    
    occupancy = db.query("""
        SELECT * FROM vw_picker_occupancy_summary 
        WHERE picker_username = %s AND work_date = %s
    """, picker_username, work_date)
    
    quality = db.query("""
        SELECT * FROM vw_picker_quality_summary 
        WHERE picker_username = %s AND work_date = %s
    """, picker_username, work_date)
    
    trends = db.query("""
        SELECT * FROM vw_picker_weekly_trends 
        WHERE picker_username = %s 
        AND work_date >= %s - INTERVAL '7 days'
    """, picker_username, work_date)
    
    # Render HTML
    html = render_template('end_of_shift_report.html', 
        speed=speed,
        occupancy=occupancy, 
        quality=quality,
        trends=trends,
        picker=picker_username,
        date=work_date
    )
    
    # Email to manager
    send_email(
        to=get_manager_email(),
        subject=f"End-of-Shift: {picker_username} — {work_date}",
        html=html
    )
    
    # Store in dashboard
    db.picker_reports.insert_one({
        'picker': picker_username,
        'date': work_date,
        'shift_id': shift_id,
        'html': html,
        'metrics': {
            'items': speed['items_completed'],
            'efficiency': speed['avg_efficiency_ratio'],
            'occupancy': occupancy['occupancy_pct'],
            'accuracy': quality['accuracy_pct']
        }
    })
    
    return {'status': 'report_generated', 'picker': picker_username}
```

## Dashboard Card (Real-Time + Historical)

```jsx
export function PickerReportCard({ pickerUsername, workDate }) {
  const [report, setReport] = useState(null);
  const [views, setViews] = useState('summary');

  useEffect(() => {
    // Fetch report for date
    fetch(`/api/report/picker/${pickerUsername}/${workDate}`)
      .then(r => r.json())
      .then(data => setReport(data));
  }, [pickerUsername, workDate]);

  if (!report) return <div>Loading...</div>;

  return (
    <div className="picker-report-card">
      {/* Quick Summary Stats */}
      <div className="quick-stats">
        <StatTile label="Items" value={report.metrics.items} target="250-300" />
        <StatTile label="Efficiency" value={report.metrics.efficiency} target=">0.90" />
        <StatTile label="Occupancy" value={report.metrics.occupancy + '%'} target="60-70%" />
        <StatTile label="Accuracy" value={report.metrics.accuracy + '%'} target=">98%" />
      </div>

      {/* Tabs: Summary / Speed / Quality / Trends */}
      <div className="tabs">
        <button onClick={() => setViews('summary')} className={views === 'summary' ? 'active' : ''}>
          Summary
        </button>
        <button onClick={() => setViews('speed')} className={views === 'speed' ? 'active' : ''}>
          Speed
        </button>
        <button onClick={() => setViews('quality')} className={views === 'quality' ? 'active' : ''}>
          Quality
        </button>
        <button onClick={() => setViews('trends')} className={views === 'trends' ? 'active' : ''}>
          Trends
        </button>
      </div>

      {/* Content by View */}
      {views === 'summary' && <SummaryView report={report} />}
      {views === 'speed' && <SpeedView report={report} />}
      {views === 'quality' && <QualityView report={report} />}
      {views === 'trends' && <TrendsView report={report} />}

      {/* Actions */}
      <div className="actions">
        <button onClick={() => downloadPDF(report)}>📥 Download PDF</button>
        <button onClick={() => emailReport(report)}>📧 Email Report</button>
      </div>
    </div>
  );
}
```
